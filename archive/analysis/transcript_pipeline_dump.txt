import json
from datetime import datetime,timezone

rate_limit = json.load(
    open(
        "generated/transcript_rate_limit.json",
        encoding="utf-8"
    )
)

pause_until = rate_limit.get(
    "global_pause_until"
)

if pause_until:

    unlock = datetime.fromisoformat(
        pause_until
    )

    if unlock > datetime.now(
        timezone.utc
    ):

        print(
            "GLOBAL_TRANSCRIPT_PAUSE"
        )

        raise SystemExit

queue = json.load(
    open(
        "generated/transcript_queue.json",
        encoding="utf-8"
    )
)

for item in queue:

    if item["status"] != "ready_for_download":
        continue

    item["status"] = "claimed"

    with open(
        "generated/current_transcript_task.json",
        "w",
        encoding="utf-8"
    ) as fp:

        json.dump(
            item,
            fp,
            indent=2
        )

    with open(
        "generated/transcript_queue.json",
        "w",
        encoding="utf-8"
    ) as fp:

        json.dump(
            queue,
            fp,
            indent=2
        )

    print(
        "claimed:",
        item["video_id"]
    )

    raise SystemExit

print(
    "nothing available"
)

import json
import sys
from pathlib import Path

if len(sys.argv) != 3:

    print(
        "usage: track_id video_id"
    )

    raise SystemExit

track_id = int(sys.argv[1])

video_id = sys.argv[2]

manifest_path = (
    "generated/track_acquisition_manifest.json"
)

manifest = json.load(
    open(
        manifest_path,
        encoding="utf-8"
    )
)

transcript_exists = (
    Path(
        f"research/youtube/transcripts/{video_id}.json"
    ).exists()
)

for track in manifest:

    if track["track_id"] != track_id:
        continue

    if transcript_exists:

        track[
            "knowledge"
        ][
            "transcript_acquired"
        ] = True

        track[
            "state"
        ] = "transcript_acquired"

    break

with open(
    manifest_path,
    "w",
    encoding="utf-8"
) as fp:

    json.dump(
        manifest,
        fp,
        indent=2
    )

print(
    "transcript_found:",
    transcript_exists
)

import json
from datetime import datetime,timedelta,timezone

from src.shared.transcript_backoff import (
    transcript_backoff_minutes
)

queue = json.load(
    open(
        "generated/transcript_queue.json",
        encoding="utf-8"
    )
)

for item in queue:

    if item["status"] != "ready_for_download":
        continue

    item["attempts"] += 1

    item["last_attempt"] = (
        datetime.now(
            timezone.utc
        ).isoformat()
    )

    delay = (
        transcript_backoff_minutes(
            item["attempts"]
        )
    )

    item["next_attempt"] = (
        datetime.now(
            timezone.utc
        )
        +
        timedelta(
            minutes=delay
        )
    ).isoformat()

    item["status"] = (
        "rate_limited"
    )

    break

with open(
    "generated/transcript_queue.json",
    "w",
    encoding="utf-8"
) as fp:

    json.dump(
        queue,
        fp,
        indent=2
    )

print(
    "updated queue"
)

    YoutubeTranscriptProvider
)

    YtDlpCaptionProvider

def get_provider(
    name
):

    if name == "youtube":

        return (
            YoutubeTranscriptProvider()
        )

    if name == "ytdlp":

        return (
            YtDlpCaptionProvider()
        )

    raise ValueError(
        name
    )

import json
from pathlib import Path

for file in Path(
    "research/youtube/transcripts"
).glob("*.json"):

    try:

        data = json.load(
            open(
                file,
                encoding="utf-8"
            )
        )

        segments = []

        for event in data.get(
            "events",
            []
        ):

            text = ""

            for seg in event.get(
                "segs",
                []
            ):

                text += seg.get(
                    "utf8",
                    ""
                )

            if not text.strip():
                continue

            segments.append(
                {
                    "start":
                        event.get(
                            "tStartMs",
                            0
                        ) / 1000,

                    "text":
                        text.strip()
                }
            )

        outfile = (
            Path(
                "research/youtube/parsed_transcripts"
            )
            /
            file.name
        )

        json.dump(
            segments,
            open(
                outfile,
                "w",
                encoding="utf-8"
            ),
            indent=2
        )

        print(
            outfile.name
        )

    except Exception:

        pass

import yt_dlp
import json

task = json.load(
    open(
        "generated/current_transcript_task.json",
        encoding="utf-8"
    )
)

video_id = task[
    "video_id"
]

with yt_dlp.YoutubeDL(
    {
        "quiet": True,
        "skip_download": True
    }
) as ydl:

    info = ydl.extract_info(
        f"https://www.youtube.com/watch?v={video_id}",
        download=False
    )

caps = (
    info.get(
        "automatic_captions",
        {}
    )
)

output = {
    "video_id":
        video_id,

    "languages":
        list(
            caps.keys()
        )
}

with open(
    f"generated/caption_capabilities_{video_id}.json",
    "w",
    encoding="utf-8"
) as fp:

    json.dump(
        output,
        fp,
        indent=2
    )

print(
    "languages:",
    len(
        output["languages"]
    )
)

def transcript_backoff_minutes(
    attempts
):

    schedule = [
        30,
        120,
        360,
        720,
        1440
    ]

    if attempts >= len(schedule):
        return schedule[-1]

    return schedule[attempts]
