import json
import sys
from datetime import datetime, timedelta

track_id = int(
    sys.argv[1]
)

manifest_path = (
    "generated/track_acquisition_manifest.json"
)

manifest = json.load(
    open(
        manifest_path,
        encoding="utf-8"
    )
)

for track in manifest:

    if track["track_id"] != track_id:
        continue

    knowledge = track[
        "knowledge"
    ]

    attempts = (
        knowledge.get(
            "transcript_attempts",
            0
        ) + 1
    )

    knowledge[
        "transcript_attempts"
    ] = attempts

    cooldown = min(
        attempts * 30,
        1440
    )

    knowledge[
        "next_transcript_try"
    ] = (
        datetime.now().astimezone()
        +
        timedelta(
            minutes=cooldown
        )
    ).isoformat()

    break

with open(
    manifest_path,
    "w",
    encoding="utf-8"
) as fp:

    json.dump(
        manifest,
        fp,
        indent=2
    )

print(
    "cooldown scheduled"
)

from pathlib import Path
from datetime import datetime, timedelta

cooldown = Path(
    "generated/transcript_cooldown.txt"
)

if cooldown.exists():

    expires = datetime.fromisoformat(
        cooldown.read_text().strip()
    )

    if expires > datetime.now().astimezone():

        print(
            "cooldown_active"
        )

        raise SystemExit

next_time = (
    datetime.now().astimezone()
    +
    timedelta(
        minutes=30
    )
)

cooldown.write_text(
    next_time.isoformat()
)

print(
    "cooldown_started"
)

FILE: src\pipelines\release_transcript_queue.py
====================================================================================================
import json
from datetime import datetime,timezone

queue = json.load(
    open(
        "generated/transcript_queue.json",
        encoding="utf-8"
    )
)

now = datetime.now(
    timezone.utc
)

released = 0

for item in queue:

    if item["status"] != "rate_limited":
        continue

    next_attempt = item.get(
        "next_attempt"
    )

    if not next_attempt:
        continue

    retry_time = datetime.fromisoformat(
        next_attempt
    )

    if retry_time <= now:

        item["status"] = (
            "ready_for_download"
        )

        released += 1

with open(
    "generated/transcript_queue.json",
    "w",
    encoding="utf-8"
) as fp:

    json.dump(
        queue,
        fp,
        indent=2
    )

print(
    "released:",
    released
)

video_id = "1Za5r0B4YvE"

    "skip_download": True,

    "writesubtitles": False,

    "writeautomaticsub": True,

    "subtitleslangs": ["en"],

    "subtitlesformat": "vtt",

    "sleep_interval_requests": 5,

    "extractor_retries": 0,

    "outtmpl": "research/youtube/transcripts/%(id)s"
}

    ydl.download(
        [url]
    )

    "caption_download_complete"
)
